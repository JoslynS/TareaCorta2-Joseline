❤️ Corazón en Python

import turtle as t
import math as m
import random as r
import time

s = t.Screen()
s.setup(700, 700)
s.bgcolor("black")
s.title("Mi corazon por ti <3")
s.tracer(0)

p = t.Turtle()
p.hideturtle()

def heart(a, scale):
    x = 16 * (m.sin(a)**3) * scale
    y = (13 * m.cos(a) - 5 * m.cos(2*a) - 2 * m.cos(3*a) - m.cos(4*a)) * scale
    return x, y

try:
    for i in range(10000):
        a = r.uniform(0, 2 * m.pi)
        sc = r.uniform(0.5, 15.5)
        x, y = heart(a, sc)
        ang = m.atan2(y, x) + r.uniform(-0.5, 0.5)
        length = r.uniform(4, 14)
        p.pencolor(1.0, r.uniform(0.25, 0.53), r.uniform(0.63, 0.85))
        p.width(r.uniform(0.3, 1.2))
        p.penup()
        p.goto(x + r.uniform(-2, 2), y + r.uniform(-2, 2))
        p.pendown()
        p.goto(x + length * m.cos(ang), y + length * m.sin(ang))
        if i % 200 == 0:
            s.update()
            time.sleep(0.002)

    for i in range(3500):
        a = r.uniform(0, 2 * m.pi)
        x, y = heart(a, 16.0)
        ang = m.atan2(y, x) + r.uniform(-0.35, 0.35)
        length = r.uniform(10, 32)
        p.pencolor(1.0, r.uniform(0.45, 0.75), r.uniform(0.75, 0.95))
        p.width(r.uniform(0.4, 0.9))
        p.penup()
        p.goto(x + r.uniform(-2, 2), y + r.uniform(-2, 2))
        p.pendown()
        p.goto(x + length * m.cos(ang), y + length * m.sin(ang))
        if i % 150 == 0:
            s.update()
            time.sleep(0.002)

    s.update()
    t.done()

except t.Terminator:
    print("Ventana cerrada. ¡Adiós!")
except Exception:
    pass

import random

secreto = random.randint(1, 20)
intentos = 0

print("Adivina un número entre 1 y 20")

while True:
    intento = int(input("Tu intento: "))
    intentos += 1

    if intento < secreto:
        print("Muy bajo. ")
    elif intento > secreto:
        print("Muy alto. ")
    else:
        print(f"Correcto, lo has logrado en {intentos} intentos, felicidades. ")
        break
